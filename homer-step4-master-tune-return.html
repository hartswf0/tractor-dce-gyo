function onSaveAndCopy(newMpdText) {
  const bus = new BroadcastChannel('wag-frank');
  bus.postMessage({
    kind: 'master-export-mpd',
    payload: { mpd_text: newMpdText }
  });
}

// In HOMER shell
bus.onmessage = (ev) => {
  if (ev.data.kind === 'master-export-mpd') {
    const mpd = ev.data.payload.mpd_text;
    updateCourageMpd(mpd);
    updateWeaverMpd(mpd);
  }
};
After this step:
1. Courage's MPD text has Master's precise positions and flips baked in.
